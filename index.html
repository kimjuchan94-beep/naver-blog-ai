import os import asyncio from pathlib import Path from PIL import Image from pillow_heif import register_heif_opener from playwright.async_api import async_playwright # 아이폰 HEIC/HEIF 포맷 지원 등록 register_heif_opener() # ========================================== # 1. 설정 항목 # ========================================== NAVER_ID = "본인의_네이버_아이디" # 본인 네이버 ID로 변경 IMAGE_DIR = "./photos" # 갤럭시/아이폰 사진을 넣은 폴더 경로 BLOG_TITLE = "노인보호전문기관 정보제공 및 비공개 지침 완벽 정리" BLOG_CONTENT = """안녕하세요! 노인보호전문기관 정보제공 지침 핵심 요약입니다. 1. 기본 원칙 노인보호전문기관의 상담일지 및 조사서 등은 개인정보 보호를 위해 비공개를 원칙으로 합니다. 2. 수사기관 및 유관기관 대응 영장 등 강제력이 있는 경우에만 제출하며, 일반 요청 시 법적 근거가 명시된 공문을 요구해야 합니다.""" # ========================================== # 2. 이미지 호환 변환 함수 (갤럭시 & 아이폰 공통) # ========================================== def prepare_images(image_directory): """갤럭시/아이폰 사진(HEIC, JPG, PNG 등)을 스캔하여 호환 가능한 JPG 목록으로 반환""" supported_extensions = {".jpg", ".jpeg", ".png", ".heic", ".heif", ".webp"} valid_image_paths = [] img_dir = Path(image_directory) if not img_dir.exists(): os.makedirs(img_dir, exist_ok=True) print(f"📁 '{image_directory}' 폴더가 생성되었습니다. 여기에 사진을 넣어주세요.") return [] for file_path in img_dir.iterdir(): ext = file_path.suffix.lower() if ext in supported_extensions: # 아이폰 HEIC/HEIF 파일인 경우 JPG로 자동 변환 if ext in {".heic", ".heif"}: try: converted_path = file_path.with_suffix(".jpg") image = Image.open(file_path) image.convert("RGB").save(converted_path, "JPEG") print(f"🔄 아이폰 HEIC 사진 자동 변환 완료: {file_path.name} -> {converted_path.name}") valid_image_paths.append(str(converted_path.resolve())) except Exception as e: print(f"⚠️ {file_path.name} 변환 실패: {e}") else: valid_image_paths.append(str(file_path.resolve())) return valid_image_paths # ========================================== # 3. 네이버 블로그 자동 임시저장 실행 함수 # ========================================== async def main(): # 1) 사진 파일 처리 및 경로 확보 images_to_upload = prepare_images(IMAGE_DIR) print(f"📸 업로드할 총 사진 개수: {len(images_to_upload)}장") async with async_playwright() as p: # 브라우저 실행 (세션 유지를 통해 매번 로그인하지 않음) context = await p.chromium.launch_persistent_context( user_data_dir="./naver_user_data", headless=False, args=["--start-maximized"] ) page = await context.new_page() # 네이버 블로그 글쓰기 접속 print("🔗 네이버 블로그 에디터 접속 중...") await page.goto(f"https://blog.naver.com/{NAVER_ID}?Redirect=Write") # 스마트에디터 iframe 전환 대기 await page.wait_for_selector("iframe#mainFrame", timeout=60000) frame = page.frame(name="mainFrame") print("💡 로그인 창이 떠 있다면 수동으로 로그인해 주세요. (자동 진행 대기 중)") # 에디터 로딩 완료 대기 await frame.wait_for_selector(".se-content", timeout=60000) await asyncio.sleep(2) # 2) 제목 입력 print("✍️ 제목 입력 중...") title_element = frame.locator(".se-document-title-textarea") await title_element.click() await title_element.fill(BLOG_TITLE) await asyncio.sleep(1) # 3) 본문 입력 print("✍️ 본문 입력 중...") content_element = frame.locator(".se-main-container") await content_element.click() await page.keyboard.type(BLOG_CONTENT, delay=30) await asyncio.sleep(1) # 4) 사진 업로드 (갤럭시 & 아이폰 사진 일괄 입력) if images_to_upload: print("🖼️ 사진 업로드 진행 중...") try: # 사진 첨부 input 태그 찾기 및 파일 할당 file_input = frame.locator("input[type='file'][accept*='image']") await file_input.set_input_files(images_to_upload) print(f"✅ 사진 {len(images_to_upload)}장 업로드 완료!") await asyncio.sleep(3) # 이미지 서버 전송 대기 except Exception as e: print(f"⚠️ 사진 자동 첨부 중 오류 발생: {e}") # 5) 임시저장 버튼 클릭 print("💾 임시저장 버튼 클릭 중...") save_button = frame.locator("button.se-save-button") await save_button.click() await asyncio.sleep(2) print("🎉 네이버 블로그에 제목, 본문, 사진 업로드 및 임시저장이 완료되었습니다!") if __name__ == "__main__": asyncio.run(main())
